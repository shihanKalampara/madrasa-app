    <script>
        const scriptURL = 'https://script.google.com/macros/s/AKfycbwTMsuHGxjTmTYgnvy_G93XTn2G3jlLgF9BB2VmqQeXyJqBsWgfiG-J4EXcsQWVfZrrjQ/exec'; // ഇവിടെ നിങ്ങളുടെ URL നൽകുക

        function switchTab(tab) {
            if(tab === 'studentTab') {
                document.getElementById('studentForm').classList.remove('hidden');
                document.getElementById('recordForm').classList.add('hidden');
                document.getElementById('btnStudentTab').className = "flex-1 py-2 font-bold text-emerald-600 border-b-2 border-emerald-600";
                document.getElementById('btnRecordTab').className = "flex-1 py-2 font-bold text-gray-500";
            } else {
                document.getElementById('studentForm').classList.add('hidden');
                document.getElementById('recordForm').classList.remove('hidden');
                document.getElementById('btnRecordTab').className = "flex-1 py-2 font-bold text-emerald-600 border-b-2 border-emerald-600";
                document.getElementById('btnStudentTab').className = "flex-1 py-2 font-bold text-gray-500";
            }
        }

        document.getElementById('studentForm').addEventListener('submit', e => {
            e.preventDefault();
            const msg = document.getElementById('statusMsg');
            msg.textContent = "സേവ് ചെയ്യുന്നുണ്ട്...";
            
            const data = {
                action: "saveStudent",
                admissionNo: document.getElementById('admissionNo').value,
                name: document.getElementById('name').value,
                sonOf: document.getElementById('sonOf').value,
                mobile: document.getElementById('mobile').value,
                className: document.getElementById('className').value,
                photoUrl: document.getElementById('photoUrl').value
            };

            fetch(scriptURL, { method: 'POST', body: JSON.stringify(data) })
                .then(res => {
                    msg.textContent = "കുട്ടിയുടെ വിവരങ്ങൾ വിജയകരമായി സേവ് ചെയ്തു!";
                    msg.className = "mt-4 text-center font-medium text-emerald-600";
                    document.getElementById('studentForm').reset();
                }).catch(err => {
                    msg.textContent = "എന്തെങ്കിലും തകരാർ സംഭവിച്ചു!";
                    msg.className = "mt-4 text-center font-medium text-red-600";
                });
        });

        document.getElementById('recordForm').addEventListener('submit', e => {
            e.preventDefault();
            const msg = document.getElementById('statusMsg');
            msg.textContent = "അപ്ഡേറ്റ് ചെയ്യുന്നുണ്ട്...";
            
            const data = {
                action: "saveRecord",
                admissionNo: document.getElementById('recAdmissionNo').value,
                fajr: document.getElementById('fajr').value,
                zuhr: document.getElementById('zuhr').value,
                asr: document.getElementById('asr').value,
                maghrib: document.getElementById('maghrib').value,
                isha: document.getElementById('isha').value,
                examMarks: document.getElementById('examMarks').value,
                taskStatus: document.getElementById('taskStatus').value
            };

            fetch(scriptURL, { method: 'POST', body: JSON.stringify(data) })
                .then(res => {
                    msg.textContent = "വിവരങ്ങൾ വിജയകരമായി അപ്ഡേറ്റ് ചെയ്തു!";
                    msg.className = "mt-4 text-center font-medium text-emerald-600";
                    document.getElementById('recordForm').reset();
                }).catch(err => {
                    msg.textContent, "എന്തെങ്കിലും തകരാർ സംഭവിച്ചു!";
                    msg.className = "mt-4 text-center font-medium text-red-600";
                });
        });
    </script>
